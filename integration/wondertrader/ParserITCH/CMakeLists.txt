CMAKE_MINIMUM_REQUIRED(VERSION 3.0.0)
PROJECT(ParserITCH LANGUAGES CXX)
SET(CMAKE_CXX_STANDARD 17)

IF(NOT QTRADER_ROOT)
	MESSAGE(FATAL_ERROR "ParserITCH requires -DQTRADER_ROOT=/absolute/path/to/Qtrader")
ENDIF()

SET(SRC
	${PROJECT_SOURCE_DIR}/ParserITCH.cpp
	${PROJECT_SOURCE_DIR}/ParserITCH.h
)

SET(LIBRARY_OUTPUT_PATH ${CMAKE_BINARY_DIR}/build_${PLATFORM}/${CMAKE_BUILD_TYPE}/bin)

FIND_PACKAGE(Threads REQUIRED)
INCLUDE_DIRECTORIES(${INCS} ${CMAKE_SOURCE_DIR}/Includes ${QTRADER_ROOT}/include ${QTRADER_ROOT})
LINK_DIRECTORIES(${LNKS})
ADD_LIBRARY(ParserITCH SHARED ${SRC})
TARGET_COMPILE_DEFINITIONS(ParserITCH PRIVATE QTRADER_EXTERNAL_WONDERTRADER=1)
TARGET_LINK_LIBRARIES(ParserITCH PRIVATE Threads::Threads)

OPTION(QTRADER_BUILD_ITCH_SMOKE "Build ParserITCH dlopen smoke test" OFF)
IF(QTRADER_BUILD_ITCH_SMOKE)
	ADD_EXECUTABLE(ParserITCHSmoke ${PROJECT_SOURCE_DIR}/ParserITCHSmoke.cpp)
	TARGET_LINK_LIBRARIES(ParserITCHSmoke PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
	SET_TARGET_PROPERTIES(ParserITCHSmoke PROPERTIES
		RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/build_${PLATFORM}/${CMAKE_BUILD_TYPE}/bin)
ENDIF()

OPTION(QTRADER_BUILD_ITCH_UFT_SMOKE
	"Build ParserITCH -> ParserAdapter -> UFT strategy smoke test" OFF)
IF(QTRADER_BUILD_ITCH_UFT_SMOKE)
	ADD_EXECUTABLE(ParserITCHUftSmoke ${PROJECT_SOURCE_DIR}/ParserITCHUftSmoke.cpp)
	TARGET_LINK_LIBRARIES(ParserITCHUftSmoke PRIVATE
		WtUftCore WTSTools WTSUtils Threads::Threads ${CMAKE_DL_LIBS}
		boost_filesystem boost_thread)
	SET_TARGET_PROPERTIES(ParserITCHUftSmoke PROPERTIES
		RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/build_${PLATFORM}/${CMAKE_BUILD_TYPE}/bin)
	ADD_CUSTOM_COMMAND(TARGET ParserITCHUftSmoke POST_BUILD
		COMMAND ${CMAKE_COMMAND} -E make_directory
			$<TARGET_FILE_DIR:ParserITCHUftSmoke>/parsers
		COMMAND ${CMAKE_COMMAND} -E copy_if_different
			$<TARGET_FILE:ParserITCH>
			$<TARGET_FILE_DIR:ParserITCHUftSmoke>/parsers/$<TARGET_FILE_NAME:ParserITCH>)

	ADD_EXECUTABLE(ParserITCHUftBench ${PROJECT_SOURCE_DIR}/ParserITCHUftBench.cpp)
	TARGET_LINK_LIBRARIES(ParserITCHUftBench PRIVATE
		WtUftCore WTSTools WTSUtils Threads::Threads ${CMAKE_DL_LIBS}
		boost_filesystem boost_thread)
	TARGET_COMPILE_DEFINITIONS(ParserITCHUftBench PRIVATE
		QTRADER_EXTERNAL_WONDERTRADER=1)
	SET_TARGET_PROPERTIES(ParserITCHUftBench PROPERTIES
		RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/build_${PLATFORM}/${CMAKE_BUILD_TYPE}/bin)

	ADD_EXECUTABLE(ParserITCHUftFormalBench
		${PROJECT_SOURCE_DIR}/ParserITCHUftFormalBench.cpp
		${PROJECT_SOURCE_DIR}/QUftOpaque.cpp)
	TARGET_INCLUDE_DIRECTORIES(ParserITCHUftFormalBench PRIVATE
		${CMAKE_SOURCE_DIR}/WTSTools ${CMAKE_SOURCE_DIR}/WtUftCore)
	TARGET_LINK_LIBRARIES(ParserITCHUftFormalBench PRIVATE
		WtUftCore WTSTools WTSUtils Threads::Threads ${CMAKE_DL_LIBS}
		boost_filesystem boost_thread)
	TARGET_COMPILE_DEFINITIONS(ParserITCHUftFormalBench PRIVATE
		QTRADER_EXTERNAL_WONDERTRADER=1)
	IF(NOT MSVC)
		TARGET_COMPILE_OPTIONS(ParserITCHUftFormalBench PRIVATE -fno-lto)
		TARGET_LINK_OPTIONS(ParserITCHUftFormalBench PRIVATE -fno-lto)
	ENDIF()
	SET_TARGET_PROPERTIES(ParserITCHUftFormalBench PROPERTIES
		RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/build_${PLATFORM}/${CMAKE_BUILD_TYPE}/bin)
ENDIF()

IF(NOT MSVC)
	SET_TARGET_PROPERTIES(ParserITCH PROPERTIES
		CXX_VISIBILITY_PRESET hidden
		C_VISIBILITY_PRESET hidden
		VISIBILITY_INLINES_HIDDEN 1
		LINK_FLAGS_RELEASE -s)
ENDIF()
