CMAKE_MINIMUM_REQUIRED(VERSION 3.0.0)

PROJECT(WtLatencyHFTBench LANGUAGES CXX)
SET(CMAKE_CXX_STANDARD 17)

SET(EXECUTABLE_OUTPUT_PATH ${CMAKE_BINARY_DIR}/build_${PLATFORM}/${CMAKE_BUILD_TYPE}/bin/WtLatencyHFTBench)

SET(SRCS
    main.cpp
    ../WtLatencyHFT/HftLatencyTool.cpp)

INCLUDE_DIRECTORIES(${INCS})
LINK_DIRECTORIES(${LNKS})

SET(LIBS
    WtCore
    WTSTools
    WTSUtils)

IF (MSVC)
    LIST(APPEND LIBS ws2_32)
ELSE(GNUCC)
    LIST(APPEND LIBS dl pthread boost_filesystem boost_thread)
    IF(WIN32)
        LIST(APPEND LIBS ws2_32 iconv)
    ENDIF()
ENDIF()

ADD_EXECUTABLE(WtLatencyHFTBench ${SRCS})
TARGET_COMPILE_DEFINITIONS(WtLatencyHFTBench PRIVATE WT_HFT_BENCH=1)
TARGET_LINK_LIBRARIES(WtLatencyHFTBench ${LIBS})

ADD_EXECUTABLE(WtLatencyHFTBenchPool ${SRCS})
TARGET_COMPILE_DEFINITIONS(WtLatencyHFTBenchPool PRIVATE
    WT_HFT_BENCH=1
    WT_POOL_SINGLE_THREAD_FAST=1)
TARGET_LINK_LIBRARIES(WtLatencyHFTBenchPool ${LIBS})

ADD_EXECUTABLE(WtSpscQueueBench SpscQueueBench.cpp)
TARGET_LINK_LIBRARIES(WtSpscQueueBench pthread)

ADD_EXECUTABLE(WtLayoutAvx2Bench LayoutAvx2Bench.cpp)
IF (MSVC)
    TARGET_COMPILE_OPTIONS(WtLayoutAvx2Bench PRIVATE /arch:AVX2)
ELSE()
    TARGET_COMPILE_OPTIONS(WtLayoutAvx2Bench PRIVATE -mavx2)
ENDIF()

ADD_EXECUTABLE(WtShmRingProtocolTest ShmRingProtocolTest.cpp)
ADD_EXECUTABLE(WtPriceTimeOrderBookTest PriceTimeOrderBookTest.cpp)
ADD_EXECUTABLE(WtLevel2Pipeline Level2Pipeline.cpp)
TARGET_LINK_LIBRARIES(WtLevel2Pipeline pthread)
ADD_EXECUTABLE(WtLevel2PipelineDense Level2Pipeline.cpp)
TARGET_COMPILE_DEFINITIONS(WtLevel2PipelineDense PRIVATE WT_DENSE_ORDERBOOK=1)
TARGET_LINK_LIBRARIES(WtLevel2PipelineDense pthread)
ADD_EXECUTABLE(WtDenseOrderBookTest DenseOrderBookTest.cpp)
ADD_EXECUTABLE(WtDenseOrderBookBench DenseOrderBookBench.cpp)

IF (NOT MSVC)
    SET_TARGET_PROPERTIES(WtLatencyHFTBench PROPERTIES LINK_FLAGS_RELEASE -s)
    SET_TARGET_PROPERTIES(WtLatencyHFTBenchPool PROPERTIES LINK_FLAGS_RELEASE -s)
ENDIF()
